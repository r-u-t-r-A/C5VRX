cmake_minimum_required(VERSION 3.16)
# c5div-rx: a reduced fork of C5VRX-5 (https://github.com/Twotoz/C5VRX)
# for digital capture of the DAC bus by a Raspberry Pi; see README.md.
if(DEFINED C5VRX5_VERSION)
    set(PROJECT_VER "${C5VRX5_VERSION}")
else()
    set(PROJECT_VER "0.1.0-dev")
endif()
if(NOT DEFINED SDKCONFIG_DEFAULTS)
    set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(c5div_rx)
